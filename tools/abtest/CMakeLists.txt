cmake_minimum_required(VERSION 3.24)
project(AB VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.4 GIT_SHALLOW ON)
FetchContent_MakeAvailable(JUCE)
FetchContent_Declare(signalsmith-stretch GIT_REPOSITORY https://github.com/Signalsmith-Audio/signalsmith-stretch.git GIT_TAG 1.4.0 GIT_SHALLOW ON)
FetchContent_MakeAvailable(signalsmith-stretch)
juce_add_console_app(ab)
target_sources(ab PRIVATE shifter_ab.cpp ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP/MPMDetector.cpp ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP/SignalsmithShifter.cpp)
target_include_directories(ab PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP ${signalsmith-stretch_SOURCE_DIR})
target_link_libraries(ab PRIVATE signalsmith-stretch juce::juce_dsp juce::juce_audio_formats)
target_compile_definitions(ab PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
juce_add_console_app(tune)
target_sources(tune PRIVATE tune_settings.cpp)
file(GLOB DSP_CPP ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP/*.cpp)
target_sources(tune PRIVATE ${DSP_CPP})
target_include_directories(tune PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP ${signalsmith-stretch_SOURCE_DIR})
target_link_libraries(tune PRIVATE signalsmith-stretch juce::juce_dsp juce::juce_audio_formats)
target_compile_definitions(tune PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
juce_add_console_app(glide)
target_sources(glide PRIVATE autoglide_ab.cpp ${DSP_CPP})
target_include_directories(glide PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP ${signalsmith-stretch_SOURCE_DIR})
target_link_libraries(glide PRIVATE signalsmith-stretch juce::juce_dsp juce::juce_audio_formats)
target_compile_definitions(glide PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
